/* ============================================================================
 * PART 1: RULE-BASED CHATBOT ENGINE
 * ============================================================================
 * Concept:
 * A rule-based bot relies on hardcoded keyword-to-response dictionaries.
 * It does not "understand" anything: it normalizes text (e.g. to lowercase)
 * and scans for predefined substrings or regular expressions.
 */

// 1. The Rule Knowledge Base: Array of trigger patterns and canned replies
const RULES = [
  {
    keywords: ["hello", "hi", "hey", "greetings"],
    reply: "Hello! I am a rule-based bot. Ask me about weather, AI, or help."
  },
  {
    keywords: ["weather", "rain", "temperature"],
    reply: "I cannot check real-time data, but I hope it's sunny where you are!"
  },
  {
    keywords: ["ai", "artificial intelligence", "what is ai"],
    reply: "AI stands for Artificial Intelligence—machines programmed to mimic human intelligence."
  },
  {
    keywords: ["help", "what can you do"],
    reply: "I match keywords like 'hello', 'weather', 'ai', 'name', or 'bye'."
  },
  {
    keywords: ["name", "who are you"],
    reply: "I'm SmartBot 1.0! I have no memory or feelings—just if-else statements."
  },
  {
    keywords: ["bye", "goodbye", "see you"],
    reply: "Goodbye! Have a great day ahead!"
  }
];

// 2. The Matching Function: Runs on every incoming user message
function getRuleBasedReply(userText) {
  // Step A: Normalize input (convert to lowercase and trim spaces)
  const cleanInput = userText.toLowerCase().trim();

  // Step B: Iterate sequentially through every registered rule
  for (const rule of RULES) {
    // Check if any keyword in this rule appears inside the clean input
    const matchedKeyword = rule.keywords.find(kw => cleanInput.includes(kw));

    if (matchedKeyword) {
      return {
        reply: rule.reply,
        matchedKeyword: matchedKeyword,
        isFallback: false
      };
    }
  }

  // Step C: Fallback when no keywords match
  // This reveals the bot's brittleness: any phrasing outside the dictionary fails!
  return {
    reply: "I didn't recognize any keywords in that. Try asking 'What is AI?' or 'Hello'.",
    matchedKeyword: null,
    isFallback: true
  };
}
        
/* ============================================================================
 * PART 2: CONNECTING TO A LANGUAGE MODEL (LLM)
 * ============================================================================
 * Concept:
 * Unlike a rule-based bot, an LLM does not use if/else trees.
 * It is a probabilistic neural network that predicts the next most likely words (tokens).
 *
 * Notice two critical design requirements:
 * 1. Stateless API: The server forgets everything between HTTP calls.
 *    To maintain memory across turns, the client must send the FULL history every time!
 * 2. System Instructions: Defines the bot's persona and guardrails before any user text.
 */

// 1. Maintain conversation state across turns in client memory
const conversationHistory = [];

// 2. Main function to call a Language Model endpoint (e.g. Gemini / Local LLM API)
async function sendToLanguageModel(userMessage, systemPrompt) {
  // Step A: Append the new user turn to our running history array
  conversationHistory.push({
    role: "user",
    content: userMessage
  });

  /*
   * Step B: Construct the API payload.
   * - 'system_instruction': Gives the model its persistent identity and safety rules.
   * - 'contents': The entire history of turns so the model knows prior context!
   * - 'temperature': Controls randomness (0.0 = deterministic, 0.9 = creative).
   */
  const payload = {
    model: "gemini-2.5-flash", // or local Ollama / Llama model
    system_instruction: {
      parts: [{ text: systemPrompt }]
    },
    contents: conversationHistory.map(turn => ({
      role: turn.role === "user" ? "user" : "model",
      parts: [{ text: turn.content }]
    })),
    generationConfig: {
      temperature: 0.7,
      maxOutputTokens: 250
    }
  };

  /*
   * Step C: Send the HTTP POST request to the model's REST API
   * (In this web app, we simulate this call locally for instant offline learning)
   */
  try {
    const response = await fetch("https://generativelanguage.googleapis.com/v1beta/models/...", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "x-goog-api-key": "YOUR_API_KEY_HERE"
      },
      body: JSON.stringify(payload)
    });

    const data = await response.json();
    const modelReplyText = data.candidates[0].content.parts[0].text;

    // Step D: Append the model's reply to history so next turn will remember it!
    conversationHistory.push({
      role: "model",
      content: modelReplyText
    });

    return modelReplyText;
  } catch (error) {
    console.error("API Error:", error);
    return "Error contacting language model.";
  }
}
